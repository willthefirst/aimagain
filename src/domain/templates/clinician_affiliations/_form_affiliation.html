{# Shared form-body macro for affiliation create + edit. Every field is optional. Org = "(Solo practice)" → `org_id` posts blank → schema coerces to None → solo affiliation row. Location / sessions left blank → NULLs on the new row; the user can fill them in via a subsequent edit. Args: action_url — POST (create) or PATCH (edit) target. method — "post" / "patch". Drives the submit-button label: create reuses `heading`; edit inherits the framework default ("Save changes") from `actions(...)`. heading — the create submit-button text. Edit ignores it (uses the framework default). cancel_url — the action cluster's Cancel link target. delete_url — optional; when set on the edit page, renders Delete in the action cluster. delete_confirm — required when `delete_url` is set. current — optional; on edit, the loaded affiliation row used to prefill field values. Create passes None. orgs — list of organizations for the picker. Required (the handler injects it via context). #} {%- from "_shared/form_fields.html" import text_field, select_field, checkbox_field, multi_select_field, entity_select_field, textarea_field, url_field, form_section with context -%} {%- from "_shared/form_copy.html" import org_picker_help -%} {%- from "_shared/forms.html" import entity_form -%} {%- from "_shared/actions.html" import actions -%} {# The affiliation holds steady-state context only — location, insurance posture, and how-to-refer. The per-announcement profile (services / session format / clients served / cost) moved onto the opening post (`posts/_form_clinician_opening.html`). `currently_accepting_new_patients` is NOT here — it's a server-managed denormalized cache toggled by the OpeningDetail lifecycle (see the schema docstring and `src/domain/models/clinician_affiliations/README.md`). `languages` is also NOT here — it's person-level on `Clinician`. #} {% macro affiliation_form(action_url, method, heading, cancel_url, orgs, delete_url=None, delete_confirm=None, current=None) -%}
{% call entity_form(action_url, method=method) %} {% call form_section("Practice") %} {{ entity_select_field("org_id", "Organization", orgs, required=false, current=(current.org_id if current else None) , blank_label="(Solo practice — no organization)", help=org_picker_help()) }} {{ text_field("location_city", "City or area", required=false, current=(current.location_city if current else None) , maxlength=120) }} {{ select_field("location_state", "State", US_STATES, current=(current.location_state if current else None) , required=false, placeholder=true) }} {{ multi_select_field("in_network_carriers", "In-network carriers", INSURANCE_CARRIERS, labels=INSURANCE_CARRIER_LABELS, current=(current.in_network_carriers if current else None) ) }} {{ checkbox_field("accepts_out_of_network", "Accepts out-of-network patients", current=(current.accepts_out_of_network if current else true) ) }} {{ checkbox_field("sliding_scale", "Offers sliding scale", current=(current.sliding_scale if current else false) ) }} {% endcall %} {% call form_section("How to refer") %} {{ url_field("website", "Website", required=false, current=(current.website if current else None) ) }} {{ textarea_field("referral_instructions", "Referral instructions", required=false, current=(current.referral_instructions if current else None) ) }} {% endcall %} {{ actions(heading if method == "post" else none, cancel_url=cancel_url, delete_url=delete_url, delete_confirm=delete_confirm) }} {% endcall %}
{%- endmacro %}
